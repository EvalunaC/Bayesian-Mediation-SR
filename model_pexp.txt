model
{
  for(i in 1:N){
    lp.s12[i] <- inprod(mat12[i,], beta12[1:p12])
    lp.s13[i] <- inprod(mat13[i,], beta13[1:p13])
    lp.s23[i] <- inprod(mat23[i,], beta23[1:p23])
    lp1[i] <- -inprod(lam12v[1:K],E1[i,1:K])*exp(lp.s12[i])-inprod(lam13v[1:K],E1[i,1:K])*exp(lp.s13[i])
    lp2[i] <- d2[i]*(1-d1[i])*(log(inprod(lam13v[1:K],W2[i,1:K]))+lp.s13[i])
    lp3[i] <- d1[i]*(log(inprod(lam12v[1:K],W1[i,1:K]))+lp.s12[i]-inprod(lam23v[1:K],E23[i,1:K])*exp(lp.s23[i]))
    lp4[i] <- d1[i]*d2[i]*(log(inprod(lam23v[1:K],W2[i,1:K]))+lp.s23[i])
    lik[i] <- exp(lp1[i]+lp2[i]+lp3[i]+lp4[i])/C
    ones[i] ~ dbern(lik[i])
  }
  for(k in 1:K){
    lam12v[k] ~ dgamma(0.01, 0.01)
    lam13v[k] ~ dgamma(0.01, 0.01)
    lam23v[k] ~ dgamma(0.01, 0.01)
  }
  for(j in 1:p12){ beta12[j] ~ dnorm(0, 0.01) }
  for(j in 1:p13){ beta13[j] ~ dnorm(0, 0.01) }
  for(j in 1:p23){ beta23[j] ~ dnorm(0, 0.01) }
}
